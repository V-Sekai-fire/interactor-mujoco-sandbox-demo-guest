cmake_minimum_required(VERSION 3.16)
project(mujoco_cnode CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LIBRISCV_DIR "" CACHE PATH "a libriscv checkout (the godot-sandbox fork's ext/libriscv)")
if(NOT LIBRISCV_DIR)
	message(FATAL_ERROR "pass -DLIBRISCV_DIR=<libriscv checkout>")
endif()
set(RISCV_EXT_A ON CACHE BOOL "" FORCE)
set(RISCV_EXT_C ON CACHE BOOL "" FORCE)
set(RISCV_EXT_V ON CACHE BOOL "" FORCE)
set(RISCV_32I OFF CACHE BOOL "" FORCE)
set(RISCV_64I ON CACHE BOOL "" FORCE)
set(RISCV_FLAT_RW_ARENA ON CACHE BOOL "" FORCE)
set(RISCV_THREADED ON CACHE BOOL "" FORCE)
add_subdirectory(${LIBRISCV_DIR}/lib ${CMAKE_BINARY_DIR}/libriscv EXCLUDE_FROM_ALL)
target_compile_definitions(riscv PUBLIC RISCV_SYSCALLS_MAX=600 RISCV_BRK_MEMORY_SIZE=0x100000)
add_executable(host_probe host_probe.cpp)
target_link_libraries(host_probe PRIVATE riscv)
set(ERL_INTERFACE_DIR "" CACHE PATH "erl_interface directory, for example <erlang>/lib/erlang/lib/erl_interface-5.8.2")
if(ERL_INTERFACE_DIR)
	add_executable(mujoco_cnode mujoco_cnode.cpp)
	target_include_directories(mujoco_cnode PRIVATE ${ERL_INTERFACE_DIR}/include)
	target_link_libraries(mujoco_cnode PRIVATE riscv ${ERL_INTERFACE_DIR}/lib/libei.a pthread)
endif()
